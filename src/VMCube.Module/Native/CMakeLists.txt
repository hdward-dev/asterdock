cmake_minimum_required(VERSION 3.18)
project(vmcube_spice C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SPICE REQUIRED IMPORTED_TARGET spice-client-glib-2.0>=0.42)
add_executable(vmcube-spice spice_client.c)
target_link_libraries(vmcube-spice PRIVATE PkgConfig::SPICE)
set_target_properties(vmcube-spice PROPERTIES C_STANDARD 11 INSTALL_RPATH "$ORIGIN;@loader_path")
if(MSVC)
  target_compile_options(vmcube-spice PRIVATE /W4)
else()
  target_compile_options(vmcube-spice PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
endif()
install(TARGETS vmcube-spice RUNTIME DESTINATION .)
